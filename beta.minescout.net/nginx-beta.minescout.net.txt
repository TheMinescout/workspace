server {
    listen 80;
    server_name beta.minescout.net;

    root /usr/share/nginx/html/beta.minescout.net;
    index index.html index.htm;

    error_page 404 /404.html;

    # Redirect legacy .html URLs to clean URLs.
    # Example: /pages.html -> /pages
    location ~ ^/(.+)\.html$ {
        return 301 /$1;
    }

    # Serve extensionless routes from their matching .html files.
    # Example: /pages -> /pages.html
    # The browser keeps the clean URL.
    location / {
        try_files $uri $uri/ $uri.html =404;
    }

    location = /404.html {
        internal;
    }
}
